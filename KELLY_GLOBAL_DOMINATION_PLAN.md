### The Daily Topic Sync
```javascript
// Global topic determination
function getTodaysTopic() {
  const dayOfYear = getDayOfYear(new Date()); // UTC
  return CURRICULUM[dayOfYear]; // Same for everyone on Earth
}

// Next class countdown
function getNextClassTime() {
  const now = new Date();
  const minutes = now.getMinutes();
  const nextClass = Math.ceil(minutes / 10) * 10;
  return nextClass - minutes; // Minutes until next class
}
```
