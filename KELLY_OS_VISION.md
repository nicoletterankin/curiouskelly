### Live Class Database Schema

```sql
-- Live class sessions
CREATE TABLE live_sessions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_date DATE NOT NULL,
  session_time TIME NOT NULL,
  topic TEXT NOT NULL,
  status TEXT DEFAULT 'scheduled', -- 'scheduled', 'live', 'ended'
  started_at TIMESTAMPTZ,
  ended_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Live classrooms (60 per session)
CREATE TABLE live_classrooms (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  session_id UUID REFERENCES live_sessions(id),
  persona_id TEXT NOT NULL,
  age_bucket TEXT NOT NULL,
  participant_count INTEGER DEFAULT 0,
  chat_enabled BOOLEAN DEFAULT TRUE,
  questions_enabled BOOLEAN DEFAULT TRUE,
  
  UNIQUE(session_id, persona_id, age_bucket)
);

-- Participants
CREATE TABLE live_participants (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  classroom_id UUID REFERENCES live_classrooms(id),
  user_id UUID REFERENCES users(id),
  joined_at TIMESTAMPTZ DEFAULT NOW(),
  left_at TIMESTAMPTZ,
  participated BOOLEAN DEFAULT FALSE -- Did they chat/ask/react?
);

-- Chat messages
CREATE TABLE live_chat (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  classroom_id UUID REFERENCES live_classrooms(id),
  user_id UUID REFERENCES users(id),
  message TEXT NOT NULL,
  is_question BOOLEAN DEFAULT FALSE,
  is_answered BOOLEAN DEFAULT FALSE,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Questions to Kelly
CREATE TABLE live_questions (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  classroom_id UUID REFERENCES live_classrooms(id),
  user_id UUID REFERENCES users(id),
  question TEXT NOT NULL,
  answer TEXT,
  answered_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
```

### Kelly is Always On

```javascript
const KellyPresence = {
  /**
   * Kelly's availability states
   */
  states: {
    LIVE: 'live',           // Live class in progress
    ASYNC: 'async',         // Between classes, lesson available
    CHAT: 'chat',           // Always-on AI chat
    PRACTICE: 'practice',   // Practice/quiz mode
    SLEEP: 'sleep'          // 2-5 AM local time (gentle reminder to rest)
  },
  
  /**
   * Get Kelly's current state
   */
  getCurrentState() {
    const now = KellyTime.now();
    const hour = now.getHours();
    
    // Check if live class is happening
    if (this.isLiveClassActive()) {
      return this.states.LIVE;
    }
    
    // Gentle sleep reminder
    if (hour >= 2 && hour < 5) {
      return this.states.SLEEP;
    }
    
    // Always available otherwise
    return this.states.ASYNC;
  },
  
  /**
   * Check if a live class is currently active
   */
  isLiveClassActive() {
    const now = KellyTime.now();
    const hour = now.getHours();
    const minute = now.getMinutes();
    
    // Live classes at 6, 9, 12, 18, 21 hours, lasting 15 minutes
    const liveHours = [6, 9, 12, 18, 21];
    
    return liveHours.some(h => hour === h && minute < 15);
  },
  
  /**
   * Get next live class time
   */
  getNextLiveClass() {
    const now = KellyTime.now();
    const hour = now.getHours();
    const liveHours = [6, 9, 12, 18, 21];
    
    for (const h of liveHours) {
      if (h > hour || (h === hour && now.getMinutes() < 15)) {
        const next = new Date(now);
        next.setHours(h, 0, 0, 0);
        return next;
      }
    }
    
    // Next day 6 AM
    const tomorrow = new Date(now);
    tomorrow.setDate(tomorrow.getDate() + 1);
    tomorrow.setHours(6, 0, 0, 0);
    return tomorrow;
  },
  
  /**
   * Get Kelly's greeting based on state
   */
  getGreeting() {
    const state = this.getCurrentState();
    const timeGreeting = KellyTime.greeting();
    
    switch (state) {
      case this.states.LIVE:
        return `${timeGreeting}! We're LIVE right now. Join the class!`;
      case this.states.SLEEP:
        return `It's late! Get some rest. I'll be here when you wake up.`;
      default:
        return `${timeGreeting}! Ready when you are.`;
    }
  }
};
```

### WebSocket Server (Simplified)

```javascript
// Using Cloudflare Durable Objects for room state

export class LiveClassroom {
  constructor(state, env) {
    this.state = state;
    this.sessions = new Map();
    this.messageHistory = [];
  }
  
  async fetch(request) {
    const url = new URL(request.url);
    
    if (url.pathname === '/websocket') {
      const [client, server] = Object.values(new WebSocketPair());
      
      server.accept();
      this.handleConnection(server);
      
      return new Response(null, { status: 101, webSocket: client });
    }
    
    return new Response('Not found', { status: 404 });
  }
  
  handleConnection(ws) {
    const sessionId = crypto.randomUUID();
    this.sessions.set(sessionId, ws);
    
    // Send recent history
    ws.send(JSON.stringify({
      type: 'history',
      messages: this.messageHistory.slice(-50)
    }));
    
    // Broadcast participant count
    this.broadcast({
      type: 'participant_count',
      count: this.sessions.size
    });
    
    ws.addEventListener('message', async (event) => {
      const data = JSON.parse(event.data);
      
      switch (data.type) {
        case 'chat':
          this.handleChat(sessionId, data);
          break;
        case 'question':
          await this.handleQuestion(sessionId, data);
          break;
        case 'reaction':
          this.handleReaction(sessionId, data);
          break;
      }
    });
    
    ws.addEventListener('close', () => {
      this.sessions.delete(sessionId);
      this.broadcast({
        type: 'participant_count',
        count: this.sessions.size
      });
    });
  }
  
  handleChat(sessionId, data) {
    const message = {
      id: crypto.randomUUID(),
      type: 'chat',
      userId: data.userId,
      userName: data.userName,
      text: data.text,
      timestamp: Date.now()
    };
    
    this.messageHistory.push(message);
    this.broadcast(message);
  }
  
  async handleQuestion(sessionId, data) {
    // AI-powered Q&A using Claude
    const answer = await this.askKelly(data.question, data.context);
    
    const response = {
      id: crypto.randomUUID(),
      type: 'answer',
      question: data.question,
      answer: answer,
      timestamp: Date.now()
    };
    
    this.broadcast(response);
  }
  
  async askKelly(question, context) {
    const response = await fetch('https://api.anthropic.com/v1/messages', {
      method: 'POST',
      headers: {
        'Content-Type': 'application/json',
        'x-api-key': this.env.ANTHROPIC_API_KEY,
        'anthropic-version': '2023-06-01'
      },
      body: JSON.stringify({
        model: 'claude-sonnet-4-20250514',
        max_tokens: 300,
        system: `You are Kelly, teaching about "${context.topic}". 
                 You are the ${context.persona} persona, speaking to ${context.ageBucket} learners.
                 Answer briefly and warmly. Max 2-3 sentences.`,
        messages: [{ role: 'user', content: question }]
      })
    });
    
    const result = await response.json();
    return result.content[0].text;
  }
  
  handleReaction(sessionId, data) {
    this.broadcast({
      type: 'reaction',
      emoji: data.emoji,
      timestamp: Date.now()
    });
  }
  
  broadcast(message) {
    const json = JSON.stringify(message);
    for (const ws of this.sessions.values()) {
      ws.send(json);
    }
  }
}
```
